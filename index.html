🚀 Quick Start 
// Initialize SDK
const Bradicoin = require('bradicoin-sdk');

const client = new Bradicoin({
    apiKey: 'your_api_key',
    apiSecret: 'your_api_secret',
    network: 'mainnet'
});

// Create payment
const payment = await client.createPayment({
    amount: 100,
    toAddress: 'BRD-XXXX...',
    callbackUrl: 'https://yoursite.com/webhook'
});

// Check balance
const balance = await client.getBalance('BrYourAddress');
console.log(balance);
